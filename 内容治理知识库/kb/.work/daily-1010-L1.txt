#!/bin/bash
# 日更 L1 批次（BASE=daily-20261010-1010）—— 闸门 → 引擎 --apply → 落章（带 --skill）→ 记账
# 闸门严格（红即停）；引擎步记录 rc、不中断（好在同一份日志里看清每个 rc）。
cd "/Users/lute/Documents/skill-test/执行skills/执行skills/skills-回归测试/知识库/内容治理" || exit 2
R="daily-20261010-1010-L1"
SID="${DSH_SESSION_ID:-session-e6e81bae-5167-478c-8311-8ab3b22cf2d1}"
gate() {
  echo; echo "### $(date '+%H:%M:%S') GATE: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  if [ $rc -ne 0 ]; then echo "### ⛔ 闸门红 ⇒ 停（不跑本层引擎） ###"; exit 9; fi
}
step() {
  echo; echo "### $(date '+%H:%M:%S') CMD: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  return 0
}

gate sh kb/kbpy l7/scripts/l1_gate.py
step sh kb/kbpy l8/extract_l1_new_resources.py "$R" --apply
step sh kb/kbpy l8/ocr_attachments.py "$R" --apply
step sh kb/kbpy l8/apply_ocr_to_l1.py "$R" "$R" --apply
step sh kb/kbpy l8/align_l1_manifest.py "$R" --skill kb-ingest-extractor --skill kb-media-extractor --apply
step sh kb/kbpy l7/scripts/kb_pipeline.py --sync --via-skill --only L1 --run "$R" \
  --invoked kb-ingest-extractor --invoked kb-media-extractor --skill-session "L1=$SID"
echo; echo "### $(date '+%H:%M:%S') L1 批次结束 ###"
