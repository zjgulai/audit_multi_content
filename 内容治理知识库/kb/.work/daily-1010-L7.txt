#!/bin/bash
# 日更 L7 批次（BASE=daily-20261010-1010）—— L7 单独跑（跨层闸门，非层主）
# layers.yaml L7 engine 清单里 workflow_check.py 带「★ 驱动循环默认跳过（无干跑档，不加 --apply 也写盘）」
#   ⇒ 日更不跑它（本行如实记录）。
# L7 无落章引擎（STAMP_SPEC 不含 L7）：kb/l7/runs/<RUN>.json 就是编排器自己的记账文件，
#   skill_invocation_audit 对它**显式豁免**（跨层统一闸门 kb-check 不是层主）。
cd "/Users/lute/Documents/skill-test/执行skills/执行skills/skills-回归测试/知识库/内容治理" || exit 2
R="daily-20261010-1010-L7"
SID="${DSH_SESSION_ID:-session-e6e81bae-5167-478c-8311-8ab3b22cf2d1}"
gate() {
  echo; echo "### $(date '+%H:%M:%S') GATE: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  if [ $rc -ne 0 ]; then echo "### ⛔ 闸门红 ⇒ 停（不跑本层引擎） ###"; exit 9; fi
}
step() {
  echo; echo "### $(date '+%H:%M:%S') CMD: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  return 0
}

gate sh kb/kbpy l7/scripts/l7_gate.py
step sh kb/kbpy l7/scripts/skills_check.py
# workflow_check.py —— layers.yaml 明写「驱动循环默认跳过」⇒ 日更不跑
echo; echo "### $(date '+%H:%M:%S') SKIP: kb/l8/workflow_check.py（layers.yaml：驱动循环默认跳过） ###"
step sh kb/kbpy .work/l7_article_source_refresh.py "$R" --apply
step sh kb/kbpy l7/scripts/kb_pipeline.py --sync --via-skill --only L7 --run "$R" \
  --invoked kb-check --skill-session "L7=$SID"
echo; echo "### $(date '+%H:%M:%S') L7 批次结束 ###"
