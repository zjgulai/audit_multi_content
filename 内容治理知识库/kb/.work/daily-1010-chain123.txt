#!/bin/bash
# 日更 L1→L2→L3 串行链（BASE=daily-20261010-1010）
# 每层脚本内部：闸门严格（红→exit 9，本链立即停），引擎步记录 rc。
cd "/Users/lute/Documents/skill-test/执行skills/执行skills/skills-回归测试/知识库/内容治理" || exit 2
for L in L1 L2 L3; do
  echo; echo "@@@@@@@@@@ $(date '+%H:%M:%S') 开始 $L @@@@@@@@@@"
  bash "kb/.work/daily-1010-${L}.txt"
  rc=$?
  echo "@@@@@@@@@@ $(date '+%H:%M:%S') $L 结束 rc=$rc @@@@@@@@@@"
  if [ $rc -ne 0 ]; then echo "⛔ $L 非零（多为闸门红）⇒ 不进下一层"; exit $rc; fi
done
echo; echo "@@@@@@@@@@ $(date '+%H:%M:%S') L1–L3 链全部完成 @@@@@@@@@@"
