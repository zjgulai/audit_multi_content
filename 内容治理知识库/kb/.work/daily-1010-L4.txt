#!/bin/bash
# 日更 L4 批次（BASE=daily-20261010-1010）—— 闸门 → 引擎 --apply → 落章 → 记账
# 已知口径（layers.yaml L4 engine 注释 + 2026-10-10 0627 轮实测）：
#   l4_ledger_repair 会 fail-closed 拒写（exit 3）；fix_l4_repromote 判据闸门不过会拒写（exit 2）；
#   l4_integrity 的 verdict=FAIL 是**本项自检器的真结论 / 已登记漂移**，不是「L4 数据坏了」。
#   三者都如实记录 rc，不阻断本层落章（实质闸门是 l4_gate.py 的 C1–C6）。
cd "/Users/lute/Documents/skill-test/执行skills/执行skills/skills-回归测试/知识库/内容治理" || exit 2
R="daily-20261010-1010-L4"
SID="${DSH_SESSION_ID:-session-e6e81bae-5167-478c-8311-8ab3b22cf2d1}"
gate() {
  echo; echo "### $(date '+%H:%M:%S') GATE: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  if [ $rc -ne 0 ]; then echo "### ⛔ 闸门红 ⇒ 停（不跑本层引擎） ###"; exit 9; fi
}
step() {
  echo; echo "### $(date '+%H:%M:%S') CMD: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  return 0
}

gate sh kb/kbpy l7/scripts/l4_gate.py
step sh kb/kbpy .work/l4_ledger_repair.py "$R" --apply
step sh kb/kbpy .work/l4_article.py "$R" --apply
step sh kb/kbpy .work/fix_l4_repromote.py --run "$R" --apply
step sh kb/kbpy .work/l4_integrity.py
step sh kb/kbpy .work/l4_report.py "$R" --skill kb-canonical-lookup --apply
step sh kb/kbpy l7/scripts/kb_pipeline.py --sync --via-skill --only L4 --run "$R" \
  --invoked kb-canonical-lookup --skill-session "L4=$SID"
echo; echo "### $(date '+%H:%M:%S') L4 批次结束 ###"
