#!/bin/bash
# 日更 L2 批次（BASE=daily-20261010-1010）—— 闸门 → 预演 → 落章档 --apply → 记账
# skill 红线 1：先跑不带 --apply 的那次，看清「将追加 N 行」再落盘。
# L2 的落章引擎**就是** l2_split_run.py 自己（STAMP_SPEC）⇒ --skill 必须传给同一次 --apply 运行。
cd "/Users/lute/Documents/skill-test/执行skills/执行skills/skills-回归测试/知识库/内容治理" || exit 2
R="daily-20261010-1010-L2"
SID="${DSH_SESSION_ID:-session-e6e81bae-5167-478c-8311-8ab3b22cf2d1}"
gate() {
  echo; echo "### $(date '+%H:%M:%S') GATE: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  if [ $rc -ne 0 ]; then echo "### ⛔ 闸门红 ⇒ 停（不跑本层引擎） ###"; exit 9; fi
}
step() {
  echo; echo "### $(date '+%H:%M:%S') CMD: $* ###"
  "$@"; local rc=$?
  echo "### rc=$rc ###"
  return 0
}

gate sh kb/kbpy l7/scripts/l2_gate.py
step sh kb/kbpy l8/l2_split_run.py "$R"
step sh kb/kbpy l8/l2_split_run.py "$R" --skill kb-ingest-extractor --apply
step sh kb/kbpy l7/scripts/kb_pipeline.py --sync --via-skill --only L2 --run "$R" \
  --invoked kb-ingest-extractor --skill-session "L2=$SID"
echo; echo "### $(date '+%H:%M:%S') L2 批次结束 ###"
